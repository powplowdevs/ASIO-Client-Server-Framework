106// Builds [4-byte big-endian length][body].
114inline std::optional<std::vector<char>> getMessageFrame(std::string_view body, bool isHeartBeat){
115 if (body.size() > MaxMessageSize || (body.size() <= 0 && !isHeartBeat)) {
116 // No logger here: protocol functions are free functions with no Logger. The caller (Connection) logs the nullopt.
117 return std::nullopt;
118 }
119
120 std::vector<char> messageFrame;
121
122 uint32_t bodySize = body.size();
123 uint32_t header = bodySize | (isHeartBeat ? (1u << 31) : 0u);
124 uint32_t netPacket = hostToNetworkLong(header);
125
126 messageFrame.resize(sizeof(netPacket) + bodySize);
127
128 std::memcpy(messageFrame.data(), &netPacket, sizeof(netPacket));
129 std::memcpy(messageFrame.data() + sizeof(netPacket), body.data(), bodySize);
130
131 return messageFrame;
132}

134// Reads the 4 header bytes and returns the payload length and heart beat bool.
140inline std::optional<std::pair<std::uint32_t, bool>> parseHeader(const std::array<char, HeaderSize>& header){
141 std::uint32_t data = 0;
142 std::uint32_t length = 0;
143 bool isHeartbeat;
144
145 std::memcpy(&data, header.data(), HeaderSize);
146 data = networkToHostLong(data);
147
148 length = (data << 1) >> 1;
149 isHeartbeat = data & (1u << 31);
150
151 if (length > MaxMessageSize || (length <= 0 && !isHeartbeat)) {
152 // No logger here: protocol functions are free functions with no Logger. The caller (Connection) logs the nullopt.
153 return std::nullopt;
154 }
155
156 return std::make_pair(length, isHeartbeat);
157}
